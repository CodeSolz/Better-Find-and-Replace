#!/bin/bash

SOURCE_DIR=$(pwd)
DEST_DIR="$SOURCE_DIR/../real-time-auto-find-and-replace-github-prod"

# Step 1: Ensure destination folder exists
if [ ! -d "$DEST_DIR/.git" ]; then
    echo "❌ Destination repo not found! Run this first:"
    echo "git clone https://github.com/CodeSolz/Better-Find-and-Replace.git ../real-time-auto-find-and-replace-github-prod"
    exit 1
fi

# Step 2: Clean destination (except .git)
find "$DEST_DIR" -mindepth 1 -not -path "$DEST_DIR/.git*" -exec rm -rf {} +

# Step 3: Copy allowed files (exclude specific folders and files)
shopt -s dotglob
for file in "$SOURCE_DIR"/*; do
    fname=$(basename "$file")
    if [[ "$fname" != "src" && "$fname" != "node_modules" && "$fname" != "vendor" && "$fname" != ".git" && "$fname" != "github-prod.sh" && "$fname" != "package copy.json" && "$fname" != "webpack.config.js" ]]; then
        cp -r "$file" "$DEST_DIR"
    fi
done

# Step 4: Commit & push from existing Git repo
cd "$DEST_DIR" || exit

if [ -n "$(git status --porcelain)" ]; then
    git add .
    git commit -m "Update modified files only (excluding src, node_modules, vendor)"
    git push origin master
else
    echo "✅ No changes to commit"
fi

cd "$SOURCE_DIR"
